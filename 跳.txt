-- ========== 跳跃功能 ==========
function performJump()
    local character = player.Character
    if not character then return end
    
    local humanoid = character:FindFirstChild("Humanoid")
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    
    if not humanoid or not rootPart then return end
    
    if humanoid.FloorMaterial ~= Enum.Material.Air then
        pcall(function()
            humanoid:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)
            humanoid.Jump = true
        end)
        
        task.wait(0.05)
        if humanoid.FloorMaterial ~= Enum.Material.Air then
            rootPart.Velocity = Vector3.new(rootPart.Velocity.X, 55, rootPart.Velocity.Z)
        end
        
        -- 波纹动画
        local ripple = Instance.new("Frame")
        ripple.Size = UDim2.new(0, 0, 0, 0)
        ripple.Position = UDim2.new(0.5, 0, 0.5, 0)
        ripple.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
        ripple.BackgroundTransparency = 0.8
        ripple.BorderSizePixel = 0
        ripple.AnchorPoint = Vector2.new(0.5, 0.5)
        ripple.Parent = jumpButton
        
        local rippleCorner = Instance.new("UICorner")
        rippleCorner.CornerRadius = UDim.new(1, 0)
        rippleCorner.Parent = ripple
        
        local tween = tweenService:Create(
            ripple,
            TweenInfo.new(0.4, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            {Size = UDim2.new(2, 0, 2, 0), BackgroundTransparency = 1}
        )
        tween:Play()
        tween.Completed:Connect(function()
            ripple:Destroy()
        end)
    end
end

-- 键盘空格跳跃
userInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == Enum.KeyCode.Space then
        performJump()
    end
end)

-- ========== 跳跃键拖动功能（动态启用） ==========
local jumpDragData = {
    dragging = false,
    dragStart = nil,
    startPos = nil,
}

function enableJumpDrag()
    -- 移除旧的连接（防止重复）
    jumpButton.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch and not CONFIG.JumpKeyFixed then
            jumpDragData.dragging = true
            jumpDragData.dragStart = input.Position
            jumpDragData.startPos = jumpButton.Position
            jumpButton.BackgroundTransparency = 0.1
            jumpButton.Size = UDim2.new(0, 70, 0, 70)
        end
    end)

    jumpButton.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch then
            jumpDragData.dragging = false
            jumpButton.BackgroundTransparency = 0.2
            jumpButton.Size = UDim2.new(0, 80, 0, 80)
        end
    end)

    jumpButton.InputChanged:Connect(function(input)
        if jumpDragData.dragging and input.UserInputType == Enum.UserInputType.Touch and not CONFIG.JumpKeyFixed then
            local delta = input.Position - jumpDragData.dragStart
            jumpButton.Position = UDim2.new(
                jumpDragData.startPos.X.Scale,
                jumpDragData.startPos.X.Offset + delta.X,
                jumpDragData.startPos.Y.Scale,
                jumpDragData.startPos.Y.Offset + delta.Y
            )
        end
    end)
end

-- ========== 穿墙功能 ==========
local noclipConnection = nil

function enableNoclip()
    if noclipConnection then return end
    
    noclipConnection = game:GetService("RunService").Heartbeat:Connect(function()
        local character = player.Character
        if not character then return end
        
        for _, part in ipairs(character:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = false
            end
        end
    end)
    
    -- 处理新加入的部件
    player.CharacterAdded:Connect(function(char)
        if CONFIG.NoclipEnabled then
            task.wait(0.1)
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.CanCollide = false
                end
            end
        end
    end)
    
    print("🧱 穿墙模式已开启")
end

function disableNoclip()
    if noclipConnection then
        noclipConnection:Disconnect()
        noclipConnection = nil
    end
    
    -- 恢复碰撞
    local character = player.Character
    if character then
        for _, part in ipairs(character:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = true
            end
        end
    end
    
    print("🧱 穿墙模式已关闭")
end

-- ========== 角色重生处理 ==========
player.CharacterAdded:Connect(function()
    task.wait(0.5)
    print("🔄 角色重生，所有功能保持")
    
    -- 如果穿墙开启，重新应用
    if CONFIG.NoclipEnabled then
        enableNoclip()
    end
end)
